"use strict";
function cleanText(value) {
  return String(value ?? "").replace(/\s+/g, " ").trim();
}
function priceOf(value) {
  const match = cleanText(value).replace(",", ".").match(/\d+(?:\.\d+)?/);
  return match ? Number(match[0]) : null;
}
function textOf(parent, selector) {
  return cleanText(parent.querySelector(selector)?.textContent);
}
function parseProgram(doc) {
  const sections = [...doc.querySelectorAll(".den")];
  if (!sections.length) throw new Error("Селектор .den нічого не знайшов");
  return sections.map(section => {
    const heading = section.querySelector("h2");
    const day = cleanText(heading?.firstChild?.textContent);
    const date = textOf(section, "h2 small");
    const films = [...section.querySelectorAll(".film")].map(film => ({
      time: textOf(film, ".cas"),
      title: textOf(film, ".nazov"),
      price: priceOf(textOf(film, ".cena"))
    })).filter(film => film.time && film.title);
    return { day, date, films };
  });
}
function render(days) {
  const target = document.querySelector("#results") ?? document.body.appendChild(document.createElement("div"));
  target.id = "results";
  target.replaceChildren();
  for (const day of days) {
    const heading = document.createElement("h2");
    heading.textContent = `${day.day} ${day.date}`;
    target.append(heading);
    for (const film of day.films) {
      const row = document.createElement("p");
      const price = film.price === null ? "Cena neuvedená" : `${film.price.toFixed(2)} €`;
      row.textContent = `${film.time} — ${film.title} — ${price}`;
      target.append(row);
    }
  }
}

} catch (error) {

}

Ціна з комою: 8,90 € → число 8.9

Пошук: price parseFloat Number comma regex eur mena

Куди вставити: У script.js після cleanText; застосуй до .price або комірки таблиці, якщо це є у виданому HTML.

match шукає перше число. Якщо його немає, повертаємо null: це відрізняється від ціни 0.

JS
function parsePrice(text) {
  const normalized = cleanText(text).replace(",", ".");
  const match = normalized.match(/\d+(?:\.\d+)?/);
  return match ? Number(match[0]) : null;
}
// parsePrice("Cena: 8,90 €") === 8.9

Алергени з назви: (1, 3, 7)

Пошук: allergens alergeny regex match split

Куди вставити: У script.js під час побудови об’єкта страви.

JS
function parseAllergens(text) {
  const match = cleanText(text).match(/\((\d+(?:\s*,\s*\d+)*)\)/);
  return match ? match[1].split(",").map(x => x.trim()) : [];
}
// "Syr (1, 3, 7)" → ["1", "3", "7"]

Запам’ятати масив у localStorage через JSON

Пошук: JSON.stringify JSON.parse localStorage array persist

Куди вставити: У script.js: читання при старті, запис після зміни масиву.

localStorage приймає лише текст. JSON.stringify перетворює масив на текст; JSON.parse повертає масив. Не зберігай тут паролі або токени.

JS
function readSavedItems() {
  try {
    const value = JSON.parse(localStorage.getItem("saved-items") ?? "[]");
    return Array.isArray(value) ? value : [];
  } catch {
    return [];
  }
}
let savedItems = readSavedItems();
function saveItems() {
  localStorage.setItem("saved-items", JSON.stringify(savedItems));
}

Коли джерело — JSON замість HTML

Пошук: fetch JSON response.json array

async function loadJson(url) {
  const response = await fetch(url);
  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  const data = await response.json();

}

Кілька джерел: Promise.allSettled

Пошук: parallel Promise.allSettled Promise.all failure 404 multiple files

Куди вставити: Тільки якщо в умові реально дано кілька адрес HTML.

Передай функції точні адреси з умови. allSettled поверне результат кожного запиту окремо; невдалий не приховає вдалі.

JS
async function loadAllSources(sourceUrls) {
  // Передай сюди ТІЛЬКИ адреси, які реально є в умові.
  const requests = sourceUrls.map(async url => {
    const response = await fetch(url);
    if (!response.ok) throw new Error(`HTTP ${response.status}`);
    return new DOMParser().parseFromString(await response.text(), "text/html");
  });
  const outcomes = await Promise.allSettled(requests);
  outcomes.forEach((outcome, index) => {
    if (outcome.status === "fulfilled") console.log(sourceUrls[index], outcome.value);
    else console.error(sourceUrls[index], outcome.reason);
  });
  return outcomes;
}
// Виклик: loadAllSources(адресиЗУмови);
